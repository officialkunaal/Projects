char* longestCommonPrefix(char** strs, int strsSize) {
    if (strsSize == 0) {
        return "";
    }

    char* first = *strs;

    for (int col = 0; *(first + col) != '\0'; col++) {
        char c = *(first + col);

        for (int row = 1; row < strsSize; row++) {
            char current = *(*(strs + row) + col);


            if (current == '\0' || current != c) {
                *(first + col) = '\0';
                return first;
            }
        }
    }

    return first;
}
