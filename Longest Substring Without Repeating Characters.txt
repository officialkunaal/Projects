int lengthOfLongestSubstring(char* s) {
    int last_seen[128];
    for (int i = 0; i < 128; i++) {
        *(last_seen + i) = -1;
    }

    int max_len = 0;
    int left = 0;

    for (int right = 0; *(s + right) != '\0'; right++) {
        unsigned char c = (unsigned char)*(s + right);

        if (*(last_seen + c) >= left) {
            left = *(last_seen + c) + 1;
        }

        *(last_seen + c) = right;

        int current_len = right - left + 1;
        if (current_len > max_len) {
            max_len = current_len;
        }
    }

    return max_len;
}
